const jwt = require('jsonwebtoken');
const db = require('./api'); // Подключение к базе данных

const selectCourse = async (req, res) => {
    try {
        const token = req.headers.authorization?.split(' ')[1];
        if (!token) {
            return res.status(401).json({ message: 'Неавторизованный пользователь' });
        }

        const decoded = jwt.verify(token, 'fcea262ab199c23d3929ecae328ad1492179fd31bc2496413c98bcc3d26de024'); // Используй свой секрет
        const userId = decoded.id;

        const { purposeId } = req.body;

        // Проверка существования курса
        const [existingCourse] = await db.query('SELECT * FROM purpose WHERE id = ?', [purposeId]);
        if (existingCourse.length === 0) {
            return res.status(404).json({ message: 'Курс не найден.' });
        }

        // Проверка, добавлен ли уже курс в журнал
        const [existingEntry] = await db.query(
            'SELECT * FROM journal WHERE user_id = ? AND purpose_config_id = ?',
            [userId, purposeId]
        );
        if (existingEntry.length > 0) {
            return res.status(400).json({ message: 'Этот курс уже выбран.' });
        }

        // Генерация плана (пример)
        for (let day = 1; day <= 30; day++) {
            const [randomExercise] = await db.query(
                'SELECT id FROM exercise WHERE strength = ? ORDER BY RAND() LIMIT 1',
                [purposeId]
            );

            if (randomExercise.length > 0) {
                await db.query(
                    'INSERT INTO purpose_config (purpose_id, exercise_id, day) VALUES (?, ?, ?)',
                    [purposeId, randomExercise[0].id, day]
                );
            }
        }

        // Добавление записи в журнал
        await db.query(
            'INSERT INTO journal (user_id, purpose_config_id, date_started) VALUES (?, ?, NULL)',
            [userId, purposeId, null]
        );

        return res.status(200).json({ message: 'Курс успешно выбран и добавлен в журнал.' });
    } catch (error) {
        console.error(error);
        return res.status(500).json({ message: 'Ошибка сервера.' });
    }
};

module.exports = {selectCourse};
